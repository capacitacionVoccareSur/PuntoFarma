<script type="module">
import * as THREE from 'three';
import { RoundedBoxGeometry } from 'three/addons/geometries/RoundedBoxGeometry.js';

function makeIcon(id, geometry, color, emissive, opts = {}) {
  const canvas = document.getElementById(id);
  if (!canvas) return;

  const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true });
  renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
  renderer.setSize(64, 64, false);

  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(42, 1, 0.1, 20);
  camera.position.z = 2.6;

  scene.add(new THREE.AmbientLight(0xffffff, 0.4));
  const sun = new THREE.DirectionalLight(0xffffff, 1.8);
  sun.position.set(3, 5, 3);
  scene.add(sun);
  const glow = new THREE.PointLight(emissive, 1.6, 7);
  glow.position.set(-1.5, -1, 1.5);
  scene.add(glow);
  const rim = new THREE.DirectionalLight(0xffffff, 0.35);
  rim.position.set(-2, 2, -2);
  scene.add(rim);

  const mat = new THREE.MeshPhysicalMaterial({
    color,
    emissive,
    emissiveIntensity: 0.1,
    metalness:  opts.metalness  ?? 0.12,
    roughness:  opts.roughness  ?? 0.05,
    clearcoat:  opts.clearcoat  ?? 1.0,
    clearcoatRoughness: 0.08,
    ...(opts.extra ?? {})
  });

  const mesh = new THREE.Mesh(geometry, mat);
  mesh.rotation.x = opts.rx ?? 0.3;
  mesh.rotation.y = opts.ry ?? 0.4;
  scene.add(mesh);

  const spd   = opts.speed ?? 0.007;
  const baseX = opts.rx ?? 0.3;
  let t = 0;

  renderer.setAnimationLoop(() => {
    t += 0.016;
    mesh.rotation.y += spd;
    mesh.rotation.x = baseX + Math.sin(t * 0.55) * 0.07;
    renderer.render(scene, camera);
  });
}
